## 📜 نص السورة (قراءة متواصلة)

يَٰٓأَيُّهَا ٱلْمُدَّثِّرُ ﴿١﴾ قُمْ فَأَنذِرْ ﴿٢﴾ وَرَبَّكَ فَكَبِّرْ ﴿٣﴾ وَثِيَابَكَ فَطَهِّرْ ﴿٤﴾ وَٱلرُّجْزَ فَٱهْجُرْ ﴿٥﴾ وَلَا تَمْنُن تَسْتَكْثِرُ ﴿٦﴾ وَلِرَبِّكَ فَٱصْبِرْ ﴿٧﴾ فَإِذَا نُقِرَ فِى ٱلنَّاقُورِ ﴿٨﴾ فَذَٰلِكَ يَوْمَئِذٍۢ يَوْمٌ عَسِيرٌ ﴿٩﴾ عَلَى ٱلْكَٰفِرِينَ غَيْرُ يَسِيرٍۢ ﴿١٠﴾ ذَرْنِى وَمَنْ خَلَقْتُ وَحِيدًۭا ﴿١١﴾ وَجَعَلْتُ لَهُۥ مَالًۭا مَّمْدُودًۭا ﴿١٢﴾ وَبَنِينَ شُهُودًۭا ﴿١٣﴾ وَمَهَّدتُّ لَهُۥ تَمْهِيدًۭا ﴿١٤﴾ ثُمَّ يَطْمَعُ أَنْ أَزِيدَ ﴿١٥﴾ كَلَّآ ۖ إِنَّهُۥ كَانَ لِءَايَٰتِنَا عَنِيدًۭا ﴿١٦﴾ سَأُرْهِقُهُۥ صَعُودًا ﴿١٧﴾ إِنَّهُۥ فَكَّرَ وَقَدَّرَ ﴿١٨﴾ فَقُتِلَ كَيْفَ قَدَّرَ ﴿١٩﴾ ثُمَّ قُتِلَ كَيْفَ قَدَّرَ ﴿٢٠﴾ ثُمَّ نَظَرَ ﴿٢١﴾ ثُمَّ عَبَسَ وَبَسَرَ ﴿٢٢﴾ ثُمَّ أَدْبَرَ وَٱسْتَكْبَرَ ﴿٢٣﴾ فَقَالَ إِنْ هَٰذَآ إِلَّا سِحْرٌۭ يُؤْثَرُ ﴿٢٤﴾ إِنْ هَٰذَآ إِلَّا قَوْلُ ٱلْبَشَرِ ﴿٢٥﴾ سَأُصْلِيهِ سَقَرَ ﴿٢٦﴾ وَمَآ أَدْرَىٰكَ مَا سَقَرُ ﴿٢٧﴾ لَا تُبْقِى وَلَا تَذَرُ ﴿٢٨﴾ لَوَّاحَةٌۭ لِّلْبَشَرِ ﴿٢٩﴾ عَلَيْهَا تِسْعَةَ عَشَرَ ﴿٣٠﴾ وَمَا جَعَلْنَآ أَصْحَٰبَ ٱلنَّارِ إِلَّا مَلَٰٓئِكَةًۭ ۙ وَمَا جَعَلْنَا عِدَّتَهُمْ إِلَّا فِتْنَةًۭ لِّلَّذِينَ كَفَرُوا۟ لِيَسْتَيْقِنَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَٰبَ وَيَزْدَادَ ٱلَّذِينَ ءَامَنُوٓا۟ إِيمَٰنًۭا ۙ وَلَا يَرْتَابَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَٰبَ وَٱلْمُؤْمِنُونَ ۙ وَلِيَقُولَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ وَٱلْكَٰفِرُونَ مَاذَآ أَرَادَ ٱللَّهُ بِهَٰذَا مَثَلًۭا ۚ كَذَٰلِكَ يُضِلُّ ٱللَّهُ مَن يَشَآءُ وَيَهْدِى مَن يَشَآءُ ۚ وَمَا يَعْلَمُ جُنُودَ رَبِّكَ إِلَّا هُوَ ۚ وَمَا هِىَ إِلَّا ذِكْرَىٰ لِلْبَشَرِ ﴿٣١﴾ كَلَّا وَٱلْقَمَرِ ﴿٣٢﴾ وَٱلَّيْلِ إِذْ أَدْبَرَ ﴿٣٣﴾ وَٱلصُّبْحِ إِذَآ أَسْفَرَ ﴿٣٤﴾ إِنَّهَا لَإِحْدَى ٱلْكُبَرِ ﴿٣٥﴾ نَذِيرًۭا لِّلْبَشَرِ ﴿٣٦﴾ لِمَن شَآءَ مِنكُمْ أَن يَتَقَدَّمَ أَوْ يَتَأَخَّرَ ﴿٣٧﴾ كُلُّ نَفْسٍۭ بِمَا كَسَبَتْ رَهِينَةٌ ﴿٣٨﴾ إِلَّآ أَصْحَٰبَ ٱلْيَمِينِ ﴿٣٩﴾ فِى جَنَّٰتٍۢ يَتَسَآءَلُونَ ﴿٤٠﴾ عَنِ ٱلْمُجْرِمِينَ ﴿٤١﴾ مَا سَلَكَكُمْ فِى سَقَرَ ﴿٤٢﴾ قَالُوا۟ لَمْ نَكُ مِنَ ٱلْمُصَلِّينَ ﴿٤٣﴾ وَلَمْ نَكُ نُطْعِمُ ٱلْمِسْكِينَ ﴿٤٤﴾ وَكُنَّا نَخُوضُ مَعَ ٱلْخَآئِضِينَ ﴿٤٥﴾ وَكُنَّا نُكَذِّبُ بِيَوْمِ ٱلدِّينِ ﴿٤٦﴾ حَتَّىٰٓ أَتَىٰنَا ٱلْيَقِينُ ﴿٤٧﴾ فَمَا تَنفَعُهُمْ شَفَٰعَةُ ٱلشَّٰفِعِينَ ﴿٤٨﴾ فَمَا لَهُمْ عَنِ ٱلتَّذْكِرَةِ مُعْرِضِينَ ﴿٤٩﴾ كَأَنَّهُمْ حُمُرٌۭ مُّسْتَنفِرَةٌۭ ﴿٥٠﴾ فَرَّتْ مِن قَسْوَرَةٍۭ ﴿٥١﴾ بَلْ يُرِيدُ كُلُّ ٱمْرِئٍۢ مِّنْهُمْ أَن يُؤْتَىٰ صُحُفًۭا مُّنَشَّرَةًۭ ﴿٥٢﴾ كَلَّا ۖ بَل لَّا يَخَافُونَ ٱلْءَاخِرَةَ ﴿٥٣﴾ كَلَّآ إِنَّهُۥ تَذْكِرَةٌۭ ﴿٥٤﴾ فَمَن شَآءَ ذَكَرَهُۥ ﴿٥٥﴾ وَمَا يَذْكُرُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ ۚ هُوَ أَهْلُ ٱلتَّقْوَىٰ وَأَهْلُ ٱلْمَغْفِرَةِ ﴿٥٦﴾

## 📑 آيات السورة بالتفصيل

1. **يَٰٓأَيُّهَا ٱلْمُدَّثِّرُ** *(الجزء: 29 | الصفحة: 575)*
2. **قُمْ فَأَنذِرْ** *(الجزء: 29 | الصفحة: 575)*
3. **وَرَبَّكَ فَكَبِّرْ** *(الجزء: 29 | الصفحة: 575)*
4. **وَثِيَابَكَ فَطَهِّرْ** *(الجزء: 29 | الصفحة: 575)*
5. **وَٱلرُّجْزَ فَٱهْجُرْ** *(الجزء: 29 | الصفحة: 575)*
6. **وَلَا تَمْنُن تَسْتَكْثِرُ** *(الجزء: 29 | الصفحة: 575)*
7. **وَلِرَبِّكَ فَٱصْبِرْ** *(الجزء: 29 | الصفحة: 575)*
8. **فَإِذَا نُقِرَ فِى ٱلنَّاقُورِ** *(الجزء: 29 | الصفحة: 575)*
9. **فَذَٰلِكَ يَوْمَئِذٍۢ يَوْمٌ عَسِيرٌ** *(الجزء: 29 | الصفحة: 575)*
10. **عَلَى ٱلْكَٰفِرِينَ غَيْرُ يَسِيرٍۢ** *(الجزء: 29 | الصفحة: 575)*
11. **ذَرْنِى وَمَنْ خَلَقْتُ وَحِيدًۭا** *(الجزء: 29 | الصفحة: 575)*
12. **وَجَعَلْتُ لَهُۥ مَالًۭا مَّمْدُودًۭا** *(الجزء: 29 | الصفحة: 575)*
13. **وَبَنِينَ شُهُودًۭا** *(الجزء: 29 | الصفحة: 575)*
14. **وَمَهَّدتُّ لَهُۥ تَمْهِيدًۭا** *(الجزء: 29 | الصفحة: 575)*
15. **ثُمَّ يَطْمَعُ أَنْ أَزِيدَ** *(الجزء: 29 | الصفحة: 575)*
16. **كَلَّآ ۖ إِنَّهُۥ كَانَ لِءَايَٰتِنَا عَنِيدًۭا** *(الجزء: 29 | الصفحة: 575)*
17. **سَأُرْهِقُهُۥ صَعُودًا** *(الجزء: 29 | الصفحة: 575)*
18. **إِنَّهُۥ فَكَّرَ وَقَدَّرَ** *(الجزء: 29 | الصفحة: 576)*
19. **فَقُتِلَ كَيْفَ قَدَّرَ** *(الجزء: 29 | الصفحة: 576)*
20. **ثُمَّ قُتِلَ كَيْفَ قَدَّرَ** *(الجزء: 29 | الصفحة: 576)*
21. **ثُمَّ نَظَرَ** *(الجزء: 29 | الصفحة: 576)*
22. **ثُمَّ عَبَسَ وَبَسَرَ** *(الجزء: 29 | الصفحة: 576)*
23. **ثُمَّ أَدْبَرَ وَٱسْتَكْبَرَ** *(الجزء: 29 | الصفحة: 576)*
24. **فَقَالَ إِنْ هَٰذَآ إِلَّا سِحْرٌۭ يُؤْثَرُ** *(الجزء: 29 | الصفحة: 576)*
25. **إِنْ هَٰذَآ إِلَّا قَوْلُ ٱلْبَشَرِ** *(الجزء: 29 | الصفحة: 576)*
26. **سَأُصْلِيهِ سَقَرَ** *(الجزء: 29 | الصفحة: 576)*
27. **وَمَآ أَدْرَىٰكَ مَا سَقَرُ** *(الجزء: 29 | الصفحة: 576)*
28. **لَا تُبْقِى وَلَا تَذَرُ** *(الجزء: 29 | الصفحة: 576)*
29. **لَوَّاحَةٌۭ لِّلْبَشَرِ** *(الجزء: 29 | الصفحة: 576)*
30. **عَلَيْهَا تِسْعَةَ عَشَرَ** *(الجزء: 29 | الصفحة: 576)*
31. **وَمَا جَعَلْنَآ أَصْحَٰبَ ٱلنَّارِ إِلَّا مَلَٰٓئِكَةًۭ ۙ وَمَا جَعَلْنَا عِدَّتَهُمْ إِلَّا فِتْنَةًۭ لِّلَّذِينَ كَفَرُوا۟ لِيَسْتَيْقِنَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَٰبَ وَيَزْدَادَ ٱلَّذِينَ ءَامَنُوٓا۟ إِيمَٰنًۭا ۙ وَلَا يَرْتَابَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَٰبَ وَٱلْمُؤْمِنُونَ ۙ وَلِيَقُولَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ وَٱلْكَٰفِرُونَ مَاذَآ أَرَادَ ٱللَّهُ بِهَٰذَا مَثَلًۭا ۚ كَذَٰلِكَ يُضِلُّ ٱللَّهُ مَن يَشَآءُ وَيَهْدِى مَن يَشَآءُ ۚ وَمَا يَعْلَمُ جُنُودَ رَبِّكَ إِلَّا هُوَ ۚ وَمَا هِىَ إِلَّا ذِكْرَىٰ لِلْبَشَرِ** *(الجزء: 29 | الصفحة: 576)*
32. **كَلَّا وَٱلْقَمَرِ** *(الجزء: 29 | الصفحة: 576)*
33. **وَٱلَّيْلِ إِذْ أَدْبَرَ** *(الجزء: 29 | الصفحة: 576)*
34. **وَٱلصُّبْحِ إِذَآ أَسْفَرَ** *(الجزء: 29 | الصفحة: 576)*
35. **إِنَّهَا لَإِحْدَى ٱلْكُبَرِ** *(الجزء: 29 | الصفحة: 576)*
36. **نَذِيرًۭا لِّلْبَشَرِ** *(الجزء: 29 | الصفحة: 576)*
37. **لِمَن شَآءَ مِنكُمْ أَن يَتَقَدَّمَ أَوْ يَتَأَخَّرَ** *(الجزء: 29 | الصفحة: 576)*
38. **كُلُّ نَفْسٍۭ بِمَا كَسَبَتْ رَهِينَةٌ** *(الجزء: 29 | الصفحة: 576)*
39. **إِلَّآ أَصْحَٰبَ ٱلْيَمِينِ** *(الجزء: 29 | الصفحة: 576)*
40. **فِى جَنَّٰتٍۢ يَتَسَآءَلُونَ** *(الجزء: 29 | الصفحة: 576)*
41. **عَنِ ٱلْمُجْرِمِينَ** *(الجزء: 29 | الصفحة: 576)*
42. **مَا سَلَكَكُمْ فِى سَقَرَ** *(الجزء: 29 | الصفحة: 576)*
43. **قَالُوا۟ لَمْ نَكُ مِنَ ٱلْمُصَلِّينَ** *(الجزء: 29 | الصفحة: 576)*
44. **وَلَمْ نَكُ نُطْعِمُ ٱلْمِسْكِينَ** *(الجزء: 29 | الصفحة: 576)*
45. **وَكُنَّا نَخُوضُ مَعَ ٱلْخَآئِضِينَ** *(الجزء: 29 | الصفحة: 576)*
46. **وَكُنَّا نُكَذِّبُ بِيَوْمِ ٱلدِّينِ** *(الجزء: 29 | الصفحة: 576)*
47. **حَتَّىٰٓ أَتَىٰنَا ٱلْيَقِينُ** *(الجزء: 29 | الصفحة: 576)*
48. **فَمَا تَنفَعُهُمْ شَفَٰعَةُ ٱلشَّٰفِعِينَ** *(الجزء: 29 | الصفحة: 577)*
49. **فَمَا لَهُمْ عَنِ ٱلتَّذْكِرَةِ مُعْرِضِينَ** *(الجزء: 29 | الصفحة: 577)*
50. **كَأَنَّهُمْ حُمُرٌۭ مُّسْتَنفِرَةٌۭ** *(الجزء: 29 | الصفحة: 577)*
51. **فَرَّتْ مِن قَسْوَرَةٍۭ** *(الجزء: 29 | الصفحة: 577)*
52. **بَلْ يُرِيدُ كُلُّ ٱمْرِئٍۢ مِّنْهُمْ أَن يُؤْتَىٰ صُحُفًۭا مُّنَشَّرَةًۭ** *(الجزء: 29 | الصفحة: 577)*
53. **كَلَّا ۖ بَل لَّا يَخَافُونَ ٱلْءَاخِرَةَ** *(الجزء: 29 | الصفحة: 577)*
54. **كَلَّآ إِنَّهُۥ تَذْكِرَةٌۭ** *(الجزء: 29 | الصفحة: 577)*
55. **فَمَن شَآءَ ذَكَرَهُۥ** *(الجزء: 29 | الصفحة: 577)*
56. **وَمَا يَذْكُرُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ ۚ هُوَ أَهْلُ ٱلتَّقْوَىٰ وَأَهْلُ ٱلْمَغْفِرَةِ** *(الجزء: 29 | الصفحة: 577)*
