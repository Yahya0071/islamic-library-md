## 📜 نص السورة (قراءة متواصلة)

وَٱلْمُرْسَلَٰتِ عُرْفًۭا ﴿١﴾ فَٱلْعَٰصِفَٰتِ عَصْفًۭا ﴿٢﴾ وَٱلنَّٰشِرَٰتِ نَشْرًۭا ﴿٣﴾ فَٱلْفَٰرِقَٰتِ فَرْقًۭا ﴿٤﴾ فَٱلْمُلْقِيَٰتِ ذِكْرًا ﴿٥﴾ عُذْرًا أَوْ نُذْرًا ﴿٦﴾ إِنَّمَا تُوعَدُونَ لَوَٰقِعٌۭ ﴿٧﴾ فَإِذَا ٱلنُّجُومُ طُمِسَتْ ﴿٨﴾ وَإِذَا ٱلسَّمَآءُ فُرِجَتْ ﴿٩﴾ وَإِذَا ٱلْجِبَالُ نُسِفَتْ ﴿١٠﴾ وَإِذَا ٱلرُّسُلُ أُقِّتَتْ ﴿١١﴾ لِأَىِّ يَوْمٍ أُجِّلَتْ ﴿١٢﴾ لِيَوْمِ ٱلْفَصْلِ ﴿١٣﴾ وَمَآ أَدْرَىٰكَ مَا يَوْمُ ٱلْفَصْلِ ﴿١٤﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿١٥﴾ أَلَمْ نُهْلِكِ ٱلْأَوَّلِينَ ﴿١٦﴾ ثُمَّ نُتْبِعُهُمُ ٱلْءَاخِرِينَ ﴿١٧﴾ كَذَٰلِكَ نَفْعَلُ بِٱلْمُجْرِمِينَ ﴿١٨﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿١٩﴾ أَلَمْ نَخْلُقكُّم مِّن مَّآءٍۢ مَّهِينٍۢ ﴿٢٠﴾ فَجَعَلْنَٰهُ فِى قَرَارٍۢ مَّكِينٍ ﴿٢١﴾ إِلَىٰ قَدَرٍۢ مَّعْلُومٍۢ ﴿٢٢﴾ فَقَدَرْنَا فَنِعْمَ ٱلْقَٰدِرُونَ ﴿٢٣﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٢٤﴾ أَلَمْ نَجْعَلِ ٱلْأَرْضَ كِفَاتًا ﴿٢٥﴾ أَحْيَآءًۭ وَأَمْوَٰتًۭا ﴿٢٦﴾ وَجَعَلْنَا فِيهَا رَوَٰسِىَ شَٰمِخَٰتٍۢ وَأَسْقَيْنَٰكُم مَّآءًۭ فُرَاتًۭا ﴿٢٧﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٢٨﴾ ٱنطَلِقُوٓا۟ إِلَىٰ مَا كُنتُم بِهِۦ تُكَذِّبُونَ ﴿٢٩﴾ ٱنطَلِقُوٓا۟ إِلَىٰ ظِلٍّۢ ذِى ثَلَٰثِ شُعَبٍۢ ﴿٣٠﴾ لَّا ظَلِيلٍۢ وَلَا يُغْنِى مِنَ ٱللَّهَبِ ﴿٣١﴾ إِنَّهَا تَرْمِى بِشَرَرٍۢ كَٱلْقَصْرِ ﴿٣٢﴾ كَأَنَّهُۥ جِمَٰلَتٌۭ صُفْرٌۭ ﴿٣٣﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٣٤﴾ هَٰذَا يَوْمُ لَا يَنطِقُونَ ﴿٣٥﴾ وَلَا يُؤْذَنُ لَهُمْ فَيَعْتَذِرُونَ ﴿٣٦﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٣٧﴾ هَٰذَا يَوْمُ ٱلْفَصْلِ ۖ جَمَعْنَٰكُمْ وَٱلْأَوَّلِينَ ﴿٣٨﴾ فَإِن كَانَ لَكُمْ كَيْدٌۭ فَكِيدُونِ ﴿٣٩﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٤٠﴾ إِنَّ ٱلْمُتَّقِينَ فِى ظِلَٰلٍۢ وَعُيُونٍۢ ﴿٤١﴾ وَفَوَٰكِهَ مِمَّا يَشْتَهُونَ ﴿٤٢﴾ كُلُوا۟ وَٱشْرَبُوا۟ هَنِيٓـًٔۢا بِمَا كُنتُمْ تَعْمَلُونَ ﴿٤٣﴾ إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿٤٤﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٤٥﴾ كُلُوا۟ وَتَمَتَّعُوا۟ قَلِيلًا إِنَّكُم مُّجْرِمُونَ ﴿٤٦﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٤٧﴾ وَإِذَا قِيلَ لَهُمُ ٱرْكَعُوا۟ لَا يَرْكَعُونَ ﴿٤٨﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿٤٩﴾ فَبِأَىِّ حَدِيثٍۭ بَعْدَهُۥ يُؤْمِنُونَ ﴿٥٠﴾

## 📑 آيات السورة بالتفصيل

1. **وَٱلْمُرْسَلَٰتِ عُرْفًۭا** *(الجزء: 29 | الصفحة: 580)*
2. **فَٱلْعَٰصِفَٰتِ عَصْفًۭا** *(الجزء: 29 | الصفحة: 580)*
3. **وَٱلنَّٰشِرَٰتِ نَشْرًۭا** *(الجزء: 29 | الصفحة: 580)*
4. **فَٱلْفَٰرِقَٰتِ فَرْقًۭا** *(الجزء: 29 | الصفحة: 580)*
5. **فَٱلْمُلْقِيَٰتِ ذِكْرًا** *(الجزء: 29 | الصفحة: 580)*
6. **عُذْرًا أَوْ نُذْرًا** *(الجزء: 29 | الصفحة: 580)*
7. **إِنَّمَا تُوعَدُونَ لَوَٰقِعٌۭ** *(الجزء: 29 | الصفحة: 580)*
8. **فَإِذَا ٱلنُّجُومُ طُمِسَتْ** *(الجزء: 29 | الصفحة: 580)*
9. **وَإِذَا ٱلسَّمَآءُ فُرِجَتْ** *(الجزء: 29 | الصفحة: 580)*
10. **وَإِذَا ٱلْجِبَالُ نُسِفَتْ** *(الجزء: 29 | الصفحة: 580)*
11. **وَإِذَا ٱلرُّسُلُ أُقِّتَتْ** *(الجزء: 29 | الصفحة: 580)*
12. **لِأَىِّ يَوْمٍ أُجِّلَتْ** *(الجزء: 29 | الصفحة: 580)*
13. **لِيَوْمِ ٱلْفَصْلِ** *(الجزء: 29 | الصفحة: 580)*
14. **وَمَآ أَدْرَىٰكَ مَا يَوْمُ ٱلْفَصْلِ** *(الجزء: 29 | الصفحة: 580)*
15. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 580)*
16. **أَلَمْ نُهْلِكِ ٱلْأَوَّلِينَ** *(الجزء: 29 | الصفحة: 580)*
17. **ثُمَّ نُتْبِعُهُمُ ٱلْءَاخِرِينَ** *(الجزء: 29 | الصفحة: 580)*
18. **كَذَٰلِكَ نَفْعَلُ بِٱلْمُجْرِمِينَ** *(الجزء: 29 | الصفحة: 580)*
19. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 580)*
20. **أَلَمْ نَخْلُقكُّم مِّن مَّآءٍۢ مَّهِينٍۢ** *(الجزء: 29 | الصفحة: 581)*
21. **فَجَعَلْنَٰهُ فِى قَرَارٍۢ مَّكِينٍ** *(الجزء: 29 | الصفحة: 581)*
22. **إِلَىٰ قَدَرٍۢ مَّعْلُومٍۢ** *(الجزء: 29 | الصفحة: 581)*
23. **فَقَدَرْنَا فَنِعْمَ ٱلْقَٰدِرُونَ** *(الجزء: 29 | الصفحة: 581)*
24. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
25. **أَلَمْ نَجْعَلِ ٱلْأَرْضَ كِفَاتًا** *(الجزء: 29 | الصفحة: 581)*
26. **أَحْيَآءًۭ وَأَمْوَٰتًۭا** *(الجزء: 29 | الصفحة: 581)*
27. **وَجَعَلْنَا فِيهَا رَوَٰسِىَ شَٰمِخَٰتٍۢ وَأَسْقَيْنَٰكُم مَّآءًۭ فُرَاتًۭا** *(الجزء: 29 | الصفحة: 581)*
28. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
29. **ٱنطَلِقُوٓا۟ إِلَىٰ مَا كُنتُم بِهِۦ تُكَذِّبُونَ** *(الجزء: 29 | الصفحة: 581)*
30. **ٱنطَلِقُوٓا۟ إِلَىٰ ظِلٍّۢ ذِى ثَلَٰثِ شُعَبٍۢ** *(الجزء: 29 | الصفحة: 581)*
31. **لَّا ظَلِيلٍۢ وَلَا يُغْنِى مِنَ ٱللَّهَبِ** *(الجزء: 29 | الصفحة: 581)*
32. **إِنَّهَا تَرْمِى بِشَرَرٍۢ كَٱلْقَصْرِ** *(الجزء: 29 | الصفحة: 581)*
33. **كَأَنَّهُۥ جِمَٰلَتٌۭ صُفْرٌۭ** *(الجزء: 29 | الصفحة: 581)*
34. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
35. **هَٰذَا يَوْمُ لَا يَنطِقُونَ** *(الجزء: 29 | الصفحة: 581)*
36. **وَلَا يُؤْذَنُ لَهُمْ فَيَعْتَذِرُونَ** *(الجزء: 29 | الصفحة: 581)*
37. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
38. **هَٰذَا يَوْمُ ٱلْفَصْلِ ۖ جَمَعْنَٰكُمْ وَٱلْأَوَّلِينَ** *(الجزء: 29 | الصفحة: 581)*
39. **فَإِن كَانَ لَكُمْ كَيْدٌۭ فَكِيدُونِ** *(الجزء: 29 | الصفحة: 581)*
40. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
41. **إِنَّ ٱلْمُتَّقِينَ فِى ظِلَٰلٍۢ وَعُيُونٍۢ** *(الجزء: 29 | الصفحة: 581)*
42. **وَفَوَٰكِهَ مِمَّا يَشْتَهُونَ** *(الجزء: 29 | الصفحة: 581)*
43. **كُلُوا۟ وَٱشْرَبُوا۟ هَنِيٓـًٔۢا بِمَا كُنتُمْ تَعْمَلُونَ** *(الجزء: 29 | الصفحة: 581)*
44. **إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ** *(الجزء: 29 | الصفحة: 581)*
45. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
46. **كُلُوا۟ وَتَمَتَّعُوا۟ قَلِيلًا إِنَّكُم مُّجْرِمُونَ** *(الجزء: 29 | الصفحة: 581)*
47. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
48. **وَإِذَا قِيلَ لَهُمُ ٱرْكَعُوا۟ لَا يَرْكَعُونَ** *(الجزء: 29 | الصفحة: 581)*
49. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 29 | الصفحة: 581)*
50. **فَبِأَىِّ حَدِيثٍۭ بَعْدَهُۥ يُؤْمِنُونَ** *(الجزء: 29 | الصفحة: 581)*
