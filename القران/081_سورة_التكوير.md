## 📜 نص السورة (قراءة متواصلة)

إِذَا ٱلشَّمْسُ كُوِّرَتْ ﴿١﴾ وَإِذَا ٱلنُّجُومُ ٱنكَدَرَتْ ﴿٢﴾ وَإِذَا ٱلْجِبَالُ سُيِّرَتْ ﴿٣﴾ وَإِذَا ٱلْعِشَارُ عُطِّلَتْ ﴿٤﴾ وَإِذَا ٱلْوُحُوشُ حُشِرَتْ ﴿٥﴾ وَإِذَا ٱلْبِحَارُ سُجِّرَتْ ﴿٦﴾ وَإِذَا ٱلنُّفُوسُ زُوِّجَتْ ﴿٧﴾ وَإِذَا ٱلْمَوْءُۥدَةُ سُئِلَتْ ﴿٨﴾ بِأَىِّ ذَنۢبٍۢ قُتِلَتْ ﴿٩﴾ وَإِذَا ٱلصُّحُفُ نُشِرَتْ ﴿١٠﴾ وَإِذَا ٱلسَّمَآءُ كُشِطَتْ ﴿١١﴾ وَإِذَا ٱلْجَحِيمُ سُعِّرَتْ ﴿١٢﴾ وَإِذَا ٱلْجَنَّةُ أُزْلِفَتْ ﴿١٣﴾ عَلِمَتْ نَفْسٌۭ مَّآ أَحْضَرَتْ ﴿١٤﴾ فَلَآ أُقْسِمُ بِٱلْخُنَّسِ ﴿١٥﴾ ٱلْجَوَارِ ٱلْكُنَّسِ ﴿١٦﴾ وَٱلَّيْلِ إِذَا عَسْعَسَ ﴿١٧﴾ وَٱلصُّبْحِ إِذَا تَنَفَّسَ ﴿١٨﴾ إِنَّهُۥ لَقَوْلُ رَسُولٍۢ كَرِيمٍۢ ﴿١٩﴾ ذِى قُوَّةٍ عِندَ ذِى ٱلْعَرْشِ مَكِينٍۢ ﴿٢٠﴾ مُّطَاعٍۢ ثَمَّ أَمِينٍۢ ﴿٢١﴾ وَمَا صَاحِبُكُم بِمَجْنُونٍۢ ﴿٢٢﴾ وَلَقَدْ رَءَاهُ بِٱلْأُفُقِ ٱلْمُبِينِ ﴿٢٣﴾ وَمَا هُوَ عَلَى ٱلْغَيْبِ بِضَنِينٍۢ ﴿٢٤﴾ وَمَا هُوَ بِقَوْلِ شَيْطَٰنٍۢ رَّجِيمٍۢ ﴿٢٥﴾ فَأَيْنَ تَذْهَبُونَ ﴿٢٦﴾ إِنْ هُوَ إِلَّا ذِكْرٌۭ لِّلْعَٰلَمِينَ ﴿٢٧﴾ لِمَن شَآءَ مِنكُمْ أَن يَسْتَقِيمَ ﴿٢٨﴾ وَمَا تَشَآءُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ رَبُّ ٱلْعَٰلَمِينَ ﴿٢٩﴾

## 📑 آيات السورة بالتفصيل

1. **إِذَا ٱلشَّمْسُ كُوِّرَتْ** *(الجزء: 30 | الصفحة: 586)*
2. **وَإِذَا ٱلنُّجُومُ ٱنكَدَرَتْ** *(الجزء: 30 | الصفحة: 586)*
3. **وَإِذَا ٱلْجِبَالُ سُيِّرَتْ** *(الجزء: 30 | الصفحة: 586)*
4. **وَإِذَا ٱلْعِشَارُ عُطِّلَتْ** *(الجزء: 30 | الصفحة: 586)*
5. **وَإِذَا ٱلْوُحُوشُ حُشِرَتْ** *(الجزء: 30 | الصفحة: 586)*
6. **وَإِذَا ٱلْبِحَارُ سُجِّرَتْ** *(الجزء: 30 | الصفحة: 586)*
7. **وَإِذَا ٱلنُّفُوسُ زُوِّجَتْ** *(الجزء: 30 | الصفحة: 586)*
8. **وَإِذَا ٱلْمَوْءُۥدَةُ سُئِلَتْ** *(الجزء: 30 | الصفحة: 586)*
9. **بِأَىِّ ذَنۢبٍۢ قُتِلَتْ** *(الجزء: 30 | الصفحة: 586)*
10. **وَإِذَا ٱلصُّحُفُ نُشِرَتْ** *(الجزء: 30 | الصفحة: 586)*
11. **وَإِذَا ٱلسَّمَآءُ كُشِطَتْ** *(الجزء: 30 | الصفحة: 586)*
12. **وَإِذَا ٱلْجَحِيمُ سُعِّرَتْ** *(الجزء: 30 | الصفحة: 586)*
13. **وَإِذَا ٱلْجَنَّةُ أُزْلِفَتْ** *(الجزء: 30 | الصفحة: 586)*
14. **عَلِمَتْ نَفْسٌۭ مَّآ أَحْضَرَتْ** *(الجزء: 30 | الصفحة: 586)*
15. **فَلَآ أُقْسِمُ بِٱلْخُنَّسِ** *(الجزء: 30 | الصفحة: 586)*
16. **ٱلْجَوَارِ ٱلْكُنَّسِ** *(الجزء: 30 | الصفحة: 586)*
17. **وَٱلَّيْلِ إِذَا عَسْعَسَ** *(الجزء: 30 | الصفحة: 586)*
18. **وَٱلصُّبْحِ إِذَا تَنَفَّسَ** *(الجزء: 30 | الصفحة: 586)*
19. **إِنَّهُۥ لَقَوْلُ رَسُولٍۢ كَرِيمٍۢ** *(الجزء: 30 | الصفحة: 586)*
20. **ذِى قُوَّةٍ عِندَ ذِى ٱلْعَرْشِ مَكِينٍۢ** *(الجزء: 30 | الصفحة: 586)*
21. **مُّطَاعٍۢ ثَمَّ أَمِينٍۢ** *(الجزء: 30 | الصفحة: 586)*
22. **وَمَا صَاحِبُكُم بِمَجْنُونٍۢ** *(الجزء: 30 | الصفحة: 586)*
23. **وَلَقَدْ رَءَاهُ بِٱلْأُفُقِ ٱلْمُبِينِ** *(الجزء: 30 | الصفحة: 586)*
24. **وَمَا هُوَ عَلَى ٱلْغَيْبِ بِضَنِينٍۢ** *(الجزء: 30 | الصفحة: 586)*
25. **وَمَا هُوَ بِقَوْلِ شَيْطَٰنٍۢ رَّجِيمٍۢ** *(الجزء: 30 | الصفحة: 586)*
26. **فَأَيْنَ تَذْهَبُونَ** *(الجزء: 30 | الصفحة: 586)*
27. **إِنْ هُوَ إِلَّا ذِكْرٌۭ لِّلْعَٰلَمِينَ** *(الجزء: 30 | الصفحة: 586)*
28. **لِمَن شَآءَ مِنكُمْ أَن يَسْتَقِيمَ** *(الجزء: 30 | الصفحة: 586)*
29. **وَمَا تَشَآءُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ رَبُّ ٱلْعَٰلَمِينَ** *(الجزء: 30 | الصفحة: 586)*
