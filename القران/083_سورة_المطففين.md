## 📜 نص السورة (قراءة متواصلة)

وَيْلٌۭ لِّلْمُطَفِّفِينَ ﴿١﴾ ٱلَّذِينَ إِذَا ٱكْتَالُوا۟ عَلَى ٱلنَّاسِ يَسْتَوْفُونَ ﴿٢﴾ وَإِذَا كَالُوهُمْ أَو وَّزَنُوهُمْ يُخْسِرُونَ ﴿٣﴾ أَلَا يَظُنُّ أُو۟لَٰٓئِكَ أَنَّهُم مَّبْعُوثُونَ ﴿٤﴾ لِيَوْمٍ عَظِيمٍۢ ﴿٥﴾ يَوْمَ يَقُومُ ٱلنَّاسُ لِرَبِّ ٱلْعَٰلَمِينَ ﴿٦﴾ كَلَّآ إِنَّ كِتَٰبَ ٱلْفُجَّارِ لَفِى سِجِّينٍۢ ﴿٧﴾ وَمَآ أَدْرَىٰكَ مَا سِجِّينٌۭ ﴿٨﴾ كِتَٰبٌۭ مَّرْقُومٌۭ ﴿٩﴾ وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ ﴿١٠﴾ ٱلَّذِينَ يُكَذِّبُونَ بِيَوْمِ ٱلدِّينِ ﴿١١﴾ وَمَا يُكَذِّبُ بِهِۦٓ إِلَّا كُلُّ مُعْتَدٍ أَثِيمٍ ﴿١٢﴾ إِذَا تُتْلَىٰ عَلَيْهِ ءَايَٰتُنَا قَالَ أَسَٰطِيرُ ٱلْأَوَّلِينَ ﴿١٣﴾ كَلَّا ۖ بَلْ ۜ رَانَ عَلَىٰ قُلُوبِهِم مَّا كَانُوا۟ يَكْسِبُونَ ﴿١٤﴾ كَلَّآ إِنَّهُمْ عَن رَّبِّهِمْ يَوْمَئِذٍۢ لَّمَحْجُوبُونَ ﴿١٥﴾ ثُمَّ إِنَّهُمْ لَصَالُوا۟ ٱلْجَحِيمِ ﴿١٦﴾ ثُمَّ يُقَالُ هَٰذَا ٱلَّذِى كُنتُم بِهِۦ تُكَذِّبُونَ ﴿١٧﴾ كَلَّآ إِنَّ كِتَٰبَ ٱلْأَبْرَارِ لَفِى عِلِّيِّينَ ﴿١٨﴾ وَمَآ أَدْرَىٰكَ مَا عِلِّيُّونَ ﴿١٩﴾ كِتَٰبٌۭ مَّرْقُومٌۭ ﴿٢٠﴾ يَشْهَدُهُ ٱلْمُقَرَّبُونَ ﴿٢١﴾ إِنَّ ٱلْأَبْرَارَ لَفِى نَعِيمٍ ﴿٢٢﴾ عَلَى ٱلْأَرَآئِكِ يَنظُرُونَ ﴿٢٣﴾ تَعْرِفُ فِى وُجُوهِهِمْ نَضْرَةَ ٱلنَّعِيمِ ﴿٢٤﴾ يُسْقَوْنَ مِن رَّحِيقٍۢ مَّخْتُومٍ ﴿٢٥﴾ خِتَٰمُهُۥ مِسْكٌۭ ۚ وَفِى ذَٰلِكَ فَلْيَتَنَافَسِ ٱلْمُتَنَٰفِسُونَ ﴿٢٦﴾ وَمِزَاجُهُۥ مِن تَسْنِيمٍ ﴿٢٧﴾ عَيْنًۭا يَشْرَبُ بِهَا ٱلْمُقَرَّبُونَ ﴿٢٨﴾ إِنَّ ٱلَّذِينَ أَجْرَمُوا۟ كَانُوا۟ مِنَ ٱلَّذِينَ ءَامَنُوا۟ يَضْحَكُونَ ﴿٢٩﴾ وَإِذَا مَرُّوا۟ بِهِمْ يَتَغَامَزُونَ ﴿٣٠﴾ وَإِذَا ٱنقَلَبُوٓا۟ إِلَىٰٓ أَهْلِهِمُ ٱنقَلَبُوا۟ فَكِهِينَ ﴿٣١﴾ وَإِذَا رَأَوْهُمْ قَالُوٓا۟ إِنَّ هَٰٓؤُلَآءِ لَضَآلُّونَ ﴿٣٢﴾ وَمَآ أُرْسِلُوا۟ عَلَيْهِمْ حَٰفِظِينَ ﴿٣٣﴾ فَٱلْيَوْمَ ٱلَّذِينَ ءَامَنُوا۟ مِنَ ٱلْكُفَّارِ يَضْحَكُونَ ﴿٣٤﴾ عَلَى ٱلْأَرَآئِكِ يَنظُرُونَ ﴿٣٥﴾ هَلْ ثُوِّبَ ٱلْكُفَّارُ مَا كَانُوا۟ يَفْعَلُونَ ﴿٣٦﴾

## 📑 آيات السورة بالتفصيل

1. **وَيْلٌۭ لِّلْمُطَفِّفِينَ** *(الجزء: 30 | الصفحة: 587)*
2. **ٱلَّذِينَ إِذَا ٱكْتَالُوا۟ عَلَى ٱلنَّاسِ يَسْتَوْفُونَ** *(الجزء: 30 | الصفحة: 587)*
3. **وَإِذَا كَالُوهُمْ أَو وَّزَنُوهُمْ يُخْسِرُونَ** *(الجزء: 30 | الصفحة: 587)*
4. **أَلَا يَظُنُّ أُو۟لَٰٓئِكَ أَنَّهُم مَّبْعُوثُونَ** *(الجزء: 30 | الصفحة: 587)*
5. **لِيَوْمٍ عَظِيمٍۢ** *(الجزء: 30 | الصفحة: 587)*
6. **يَوْمَ يَقُومُ ٱلنَّاسُ لِرَبِّ ٱلْعَٰلَمِينَ** *(الجزء: 30 | الصفحة: 587)*
7. **كَلَّآ إِنَّ كِتَٰبَ ٱلْفُجَّارِ لَفِى سِجِّينٍۢ** *(الجزء: 30 | الصفحة: 588)*
8. **وَمَآ أَدْرَىٰكَ مَا سِجِّينٌۭ** *(الجزء: 30 | الصفحة: 588)*
9. **كِتَٰبٌۭ مَّرْقُومٌۭ** *(الجزء: 30 | الصفحة: 588)*
10. **وَيْلٌۭ يَوْمَئِذٍۢ لِّلْمُكَذِّبِينَ** *(الجزء: 30 | الصفحة: 588)*
11. **ٱلَّذِينَ يُكَذِّبُونَ بِيَوْمِ ٱلدِّينِ** *(الجزء: 30 | الصفحة: 588)*
12. **وَمَا يُكَذِّبُ بِهِۦٓ إِلَّا كُلُّ مُعْتَدٍ أَثِيمٍ** *(الجزء: 30 | الصفحة: 588)*
13. **إِذَا تُتْلَىٰ عَلَيْهِ ءَايَٰتُنَا قَالَ أَسَٰطِيرُ ٱلْأَوَّلِينَ** *(الجزء: 30 | الصفحة: 588)*
14. **كَلَّا ۖ بَلْ ۜ رَانَ عَلَىٰ قُلُوبِهِم مَّا كَانُوا۟ يَكْسِبُونَ** *(الجزء: 30 | الصفحة: 588)*
15. **كَلَّآ إِنَّهُمْ عَن رَّبِّهِمْ يَوْمَئِذٍۢ لَّمَحْجُوبُونَ** *(الجزء: 30 | الصفحة: 588)*
16. **ثُمَّ إِنَّهُمْ لَصَالُوا۟ ٱلْجَحِيمِ** *(الجزء: 30 | الصفحة: 588)*
17. **ثُمَّ يُقَالُ هَٰذَا ٱلَّذِى كُنتُم بِهِۦ تُكَذِّبُونَ** *(الجزء: 30 | الصفحة: 588)*
18. **كَلَّآ إِنَّ كِتَٰبَ ٱلْأَبْرَارِ لَفِى عِلِّيِّينَ** *(الجزء: 30 | الصفحة: 588)*
19. **وَمَآ أَدْرَىٰكَ مَا عِلِّيُّونَ** *(الجزء: 30 | الصفحة: 588)*
20. **كِتَٰبٌۭ مَّرْقُومٌۭ** *(الجزء: 30 | الصفحة: 588)*
21. **يَشْهَدُهُ ٱلْمُقَرَّبُونَ** *(الجزء: 30 | الصفحة: 588)*
22. **إِنَّ ٱلْأَبْرَارَ لَفِى نَعِيمٍ** *(الجزء: 30 | الصفحة: 588)*
23. **عَلَى ٱلْأَرَآئِكِ يَنظُرُونَ** *(الجزء: 30 | الصفحة: 588)*
24. **تَعْرِفُ فِى وُجُوهِهِمْ نَضْرَةَ ٱلنَّعِيمِ** *(الجزء: 30 | الصفحة: 588)*
25. **يُسْقَوْنَ مِن رَّحِيقٍۢ مَّخْتُومٍ** *(الجزء: 30 | الصفحة: 588)*
26. **خِتَٰمُهُۥ مِسْكٌۭ ۚ وَفِى ذَٰلِكَ فَلْيَتَنَافَسِ ٱلْمُتَنَٰفِسُونَ** *(الجزء: 30 | الصفحة: 588)*
27. **وَمِزَاجُهُۥ مِن تَسْنِيمٍ** *(الجزء: 30 | الصفحة: 588)*
28. **عَيْنًۭا يَشْرَبُ بِهَا ٱلْمُقَرَّبُونَ** *(الجزء: 30 | الصفحة: 588)*
29. **إِنَّ ٱلَّذِينَ أَجْرَمُوا۟ كَانُوا۟ مِنَ ٱلَّذِينَ ءَامَنُوا۟ يَضْحَكُونَ** *(الجزء: 30 | الصفحة: 588)*
30. **وَإِذَا مَرُّوا۟ بِهِمْ يَتَغَامَزُونَ** *(الجزء: 30 | الصفحة: 588)*
31. **وَإِذَا ٱنقَلَبُوٓا۟ إِلَىٰٓ أَهْلِهِمُ ٱنقَلَبُوا۟ فَكِهِينَ** *(الجزء: 30 | الصفحة: 588)*
32. **وَإِذَا رَأَوْهُمْ قَالُوٓا۟ إِنَّ هَٰٓؤُلَآءِ لَضَآلُّونَ** *(الجزء: 30 | الصفحة: 588)*
33. **وَمَآ أُرْسِلُوا۟ عَلَيْهِمْ حَٰفِظِينَ** *(الجزء: 30 | الصفحة: 588)*
34. **فَٱلْيَوْمَ ٱلَّذِينَ ءَامَنُوا۟ مِنَ ٱلْكُفَّارِ يَضْحَكُونَ** *(الجزء: 30 | الصفحة: 588)*
35. **عَلَى ٱلْأَرَآئِكِ يَنظُرُونَ** *(الجزء: 30 | الصفحة: 589)*
36. **هَلْ ثُوِّبَ ٱلْكُفَّارُ مَا كَانُوا۟ يَفْعَلُونَ** *(الجزء: 30 | الصفحة: 589)*
